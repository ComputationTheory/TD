\begin{exercice}[Grammaires engendrant des langages algébriques]\label{exo:parsing/grammars/write_grammars}

  Donnez une grammaire algébrique engendrant chacun des langages suivants :
  \begin{question}
  \item $\{ a^n b^{2n} \mid  n > 0\}$
  \item $\{x c x^\textsc{r} \mid  x \in \{a, b\}^\star  \}$ ($x^\textsc{r}$ est le mot miroir de $x$)
  \item $\{ a^n b^m \mid  n, m \in \mathbb{N}, n \neq m \}$
  \item $\{u \in \{a, b\}^\star  \mid  |u|_a = |u|_b \}$
  \end{question}

\end{exercice}

\begin{correction}

  \begin{question}
  \item $\left\langle \{a, b\}, \{S\}, S,
    \left\{\begin{array}{lll}
    S   &\rightarrow& a S bb \mid a bb\\
    \end{array}\right\}\right\rangle$
  \item $\left\langle \{a, b, c\}, \{S\}, S,
    \left\{\begin{array}{lll}
    S   &\rightarrow& a S a \mid b S b \mid c\\
    \end{array}\right\}\right\rangle$
  \item $\left\langle \{a, b\}, \{S, X, Y\}, S,
    \left\{\begin{array}{lll}
    S   &\rightarrow& X \mid Y\\
    X   &\rightarrow& a X b \mid a X \mid a\\
    Y   &\rightarrow& a Y b \mid Y b \mid b\\
    \end{array}\right\}\right\rangle$
  \item $\left\langle \{a, b\}, \{S\}, S,
    \left\{\begin{array}{lll}
    S   &\rightarrow& a S b S \mid b S a S \mid \varepsilon
    \end{array}\right\}\right\rangle$
  \end{question}

\end{correction}
