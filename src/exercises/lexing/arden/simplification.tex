\begin{exercice}[Simplification d'expressions rationnelles] \label{exo:lexing/arden/simplification}

  Proposer quelques mots vérifiant les expressions rationnelles suivantes, puis essayer de les simplifier.
  \begin{question}
  \item $((a \mid  b)^\star  c^\star )^\star $
  \item $a^\star (b^\star  \mid  c^\star )(b^\star \mid  c^\star )(b^\star  \mid  c^\star )$
  \item $(aab \mid  aac \mid  abb \mid  abc \mid  bab \mid  bac \mid  bbb \mid  bbc \mid  cab \mid  cac \mid  cbb \mid  cbc)^\star $
  \end{question}

\end{exercice}

\begin{correction}

  \begin{question}

  \item mots : $aabc$ et $cab$ ; simplification : $(a \mid  b \mid  c)^\star $
    
  \item mots : $abcb$ et $ccbc$ ;
    
    simplification :
    
    $\begin{array}{rcll}
      && a^\star (b^\star  |  c^\star )(b^\star |  c^\star )(b^\star  |  c^\star ) \\
      & \equiv & a^\star( b^\star b^\star b^\star |  b^\star b^\star c^\star |  b^\star c^\star b^\star |  b^\star c^\star c^\star  |  c^\star b^\star b^\star |  c^\star b^\star c^\star |  c^\star c^\star b^\star |  c^\star c^\star c^\star) & \text{par distribution}\\ 
      & \equiv & a^\star (b^\star |  b^\star c^\star |  b^\star c^\star b^\star |  c^\star b^\star |  c^\star b^\star c^\star |  c^\star) & L^\star\cdot L^\star  \equiv L^\star\\ 
      & \equiv & a^\star (b^\star c^\star b^\star |  c^\star b^\star c^\star) & \text{par inclusion}\\ 
    \end{array}$

  \item mots : $aabaac$ et $abccab$ ; simplification : $((a \mid  b \mid  c)(a \mid  b)(b \mid  c))^\star $ par factorisation
    
  \end{question}

\end{correction}
