\begin{exercice}[Lemme de l'étoile] \label{exo:lexing/expressiveness/pumping}

  Démontrez que les langages suivants sur l'alphabet $\Sigma = \{a, b, c\}$ \emph{ne sont pas rationnels} en utilisant le lemme de l'étoile.
  \begin{question}
  \item $\left\{u \in \Sigma^\star \mid |u|_a = |u|_b + |u|_c \right\}$
  \item $\left\{a^m b^n c^p \mid m = n \lor n = p \right\}$
  \end{question}

\end{exercice}

\begin{correction}

  \begin{question}

  \item Supposons (par l'absurde), que $L_1 \eqdef \{u \in \Sigma^\star \mid |u|_a = |u|_b + |u|_c \}$ est rationnel.
    
    Alors $L_1$ vérifie le lemme de l'étoile, donc il existe $N \in \mathbb{N}$ tel que pour tout mot $u$ tel que $|u|\ge N$,
    il existe $x, y, z \in \Sigma^\star$ tels que $u=xyz$, $y\neq \emptyset$, $|xy| \le N$ et pour tout $i\in \mathbb{N}$, $xy^i z\in L_1$.
    
    \textbf{Posons $u=a^Nb^N$.} Clairement, $u\in L_1$ (car $|u|_a=N$ et $|u|_b+|u|_c=N+0$), et $|u|=2N\ge N$.
    Comme $|xy|\le N$, $xy$ n'est composé que de $a$, donc $x=a^{|x|}$, $y=a^{|y|}\neq\varepsilon$ et $z=a^{N-|x|-|y|}b^N$.
    
    \textbf{Posons $i=0$.} On a alors $xy^iz=a^{N-|y|}b^N$.
    Or $|xy^iz|_a = N-|y| \neq N = |xy^iz|_b + |xy^iz|_c$. Donc $xy^iz \notin L_1$.
    Absurde, donc $L_1$ n'est pas rationnel.

  \item Supposons (par l'absurde), que $L_2 \eqdef \{a^m b^n c^p \mid m=n \lor n=p\}$ est rationnel.
    
    Alors $L_2$ vérifie le lemme de l'étoile, donc il existe $N \in \mathbb{N}$ tel que pour tout mot $u$ tel que $|u|\ge N$,
    il existe $x, y, z \in \Sigma^\star$ tels que $u=xyz$, $y\neq \emptyset$, $|xy| \le N$ et pour tout $i\in \mathbb{N}$, $xy^i z\in L_2$.
    
    \textbf{Posons $u=a^N b^N c^{N+1}$.} Clairement, $u\in L_2$ (car $m=n=N$), et $|u|=3N+1\ge N$.
    Comme $|xy|\le N$, $xy$ n'est composé que de $a$, donc $x=a^{|x|}$, $y=a^{|y|}\neq\varepsilon$ et $z=a^{N-|x|-|y|}b^N c^{N+1}$.
    
    \textbf{Posons $i=0$.} On a alors $xy^iz=a^{N-|y|}b^N c^{N+1} = a^{m'}b^{n'}c^{p'}$ avec $m'=N-|y|$, $n'=N$, $p'=N+1$.
    Alors $m'\neq n'$ (car $|y|>0$) et $n'\neq p'$ (car $N\neq N+1$). Donc $xy^iz \notin L_2$.
    Absurde, donc $L_2$ n'est pas rationnel.

  \end{question}

\end{correction}
